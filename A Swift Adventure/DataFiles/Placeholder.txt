
// MARK: - SliderDemo.swift
import UIKit
import SwiftUI

struct SliderDemo: WidgetDemo {
    var title: String { "Slider Demo" }
    var instructions: String { "Use the slider to select a value between 0 and 100." }

    func makeUIKitView() -> UIView {
        let slider = UISlider()
        slider.minimumValue = 0
        slider.maximumValue = 100
        slider.value = 50
        return slider
    }

    func makeSwiftUIView() -> AnyView {
        @State var value: Double = 50
        return AnyView(
            VStack {
                Slider(value: $value, in: 0...100)
                Text("Value: \(Int(value))")
            }
            .padding()
        )
    }
}

// MARK: - SegmentedControlDemo.swift
struct SegmentedControlDemo: WidgetDemo {
    var title: String { "Segmented Control Demo" }
    var instructions: String { "Tap a segment to select an option." }

    func makeUIKitView() -> UIView {
        let control = UISegmentedControl(items: ["One", "Two", "Three"])
        control.selectedSegmentIndex = 0
        return control
    }

    func makeSwiftUIView() -> AnyView {
        @State var selection = 0
        return AnyView(
            Picker("Options", selection: $selection) {
                Text("One").tag(0)
                Text("Two").tag(1)
                Text("Three").tag(2)
            }
            .pickerStyle(.segmented)
            .padding()
        )
    }
}
